# مرحله 1: اضافه کردن Output Chain Rules

/ip firewall filter add action=accept chain=output connection-state=established,related place-before=0 comment="Allow output established"

# =====================================================
# مرحله 2: اضافه کردن Forward Rule برای VPN
# =====================================================

/ip firewall filter add action=accept chain=forward out-interface=wireguard-vpn place-before=0 comment="Allow all to VPN"
