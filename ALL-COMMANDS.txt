# =====================================================
# تمام دستورات کانفیگ MikroTik VPN Server
# =====================================================
# کل این فایل را کپی کنید و در MikroTik Terminal پیست کنید
# یا فایل را به MikroTik آپلود کنید و import کنید
# =====================================================

# =====================================================
# 1. INTERFACE CONFIGURATION
# =====================================================

/interface ethernet set [ find default-name=ether1 ] name=ether1-wan-radio comment="WAN: Radio Link to Intranet"
/interface ethernet set [ find default-name=ether2 ] name=ether2-wan-vdsl comment="WAN Backup: VDSL to Intranet"
/interface ethernet set [ find default-name=ether3 ] name=ether3-lan comment="LAN Port"
/interface ethernet set [ find default-name=ether4 ] name=ether4-lan comment="LAN Port"
/interface ethernet set [ find default-name=ether5 ] name=ether5-internet comment="Internet Connection"

# =====================================================
# 2. VLAN CONFIGURATION
# =====================================================

/interface vlan add interface=ether1-wan-radio name=vlan1787 vlan-id=1787 comment="VLAN for PPPoE"

# =====================================================
# 3. PPPoE CLIENT CONFIGURATION
# =====================================================

/interface pppoe-client add add-default-route=yes default-route-distance=1 disabled=no interface=vlan1787 name=pppoe-intranet user=1331626494 password=351598 use-peer-dns=yes keepalive-timeout=60 comment="Primary WAN: Intranet via Radio Link"

# =====================================================
# 4. DHCP CLIENT CONFIGURATION
# =====================================================

/ip dhcp-client add disabled=no interface=ether2-wan-vdsl add-default-route=yes default-route-distance=2 use-peer-dns=no comment="Backup WAN: VDSL Intranet"
/ip dhcp-client add disabled=no interface=ether5-internet add-default-route=yes default-route-distance=3 use-peer-dns=yes comment="Internet Connection"

# =====================================================
# 5. LAN BRIDGE CONFIGURATION
# =====================================================

/interface bridge add name=bridge-lan comment="LAN Bridge"
/interface bridge port add bridge=bridge-lan interface=ether3-lan
/interface bridge port add bridge=bridge-lan interface=ether4-lan

# =====================================================
# 6. IP ADDRESS CONFIGURATION
# =====================================================

/ip address add address=192.168.100.1/24 interface=bridge-lan comment="LAN Network"

# =====================================================
# 7. DHCP SERVER FOR LAN
# =====================================================

/ip pool add name=lan-pool ranges=192.168.100.10-192.168.100.254
/ip dhcp-server add address-pool=lan-pool disabled=no interface=bridge-lan name=dhcp-lan
/ip dhcp-server network add address=192.168.100.0/24 gateway=192.168.100.1 dns-server=192.168.100.1,8.8.8.8,1.1.1.1 comment="LAN DHCP Network"

# =====================================================
# 8. WIREGUARD VPN SERVER CONFIGURATION
# =====================================================

/interface wireguard add listen-port=13231 mtu=1420 name=wireguard-vpn private-key="KEODtvSQS68bLhQLdJ8jmwdW7vxwCroQuVVv3hAzy3k=" comment="WireGuard VPN Server"
/ip address add address=192.168.200.1/24 interface=wireguard-vpn comment="WireGuard Server IP"
/interface wireguard peers add allowed-address=192.168.200.2/32 interface=wireguard-vpn public-key="HRnO/GFu1WFeYWkmsU65riEUPJrDvWnU//Q69bTKmQA=" comment="WireGuard Client 1"

# =====================================================
# 9. NAT CONFIGURATION
# =====================================================

/ip firewall nat add action=masquerade chain=srcnat out-interface=ether5-internet comment="NAT: LAN to Internet"
/ip firewall nat add action=masquerade chain=srcnat src-address=192.168.200.0/24 out-interface=ether5-internet comment="NAT: VPN Clients to Internet"
/ip firewall nat add action=masquerade chain=srcnat out-interface=pppoe-intranet comment="NAT: Through PPPoE"
/ip firewall nat add action=masquerade chain=srcnat out-interface=ether2-wan-vdsl comment="NAT: Through VDSL Backup"

# =====================================================
# 10. ROUTING CONFIGURATION
# =====================================================

/ip route add dst-address=0.0.0.0/0 gateway=ether5-internet distance=3 comment="Default route via Internet"

# =====================================================
# 11. FIREWALL CONFIGURATION - INPUT CHAIN
# =====================================================

/ip firewall filter add action=accept chain=input connection-state=established,related comment="Accept established/related"
/ip firewall filter add action=accept chain=input protocol=udp dst-port=13231 comment="Allow WireGuard VPN"
/ip firewall filter add action=accept chain=input protocol=icmp comment="Allow ICMP"
/ip firewall filter add action=accept chain=input in-interface=bridge-lan comment="Allow from LAN"
/ip firewall filter add action=accept chain=input in-interface=wireguard-vpn comment="Allow from VPN"
/ip firewall filter add action=drop chain=input comment="Drop all other input"

# =====================================================
# 12. FIREWALL CONFIGURATION - OUTPUT CHAIN
# =====================================================

/ip firewall filter add action=accept chain=output out-interface=wireguard-vpn comment="Allow output to VPN interface"
/ip firewall filter add action=accept chain=output connection-state=established,related comment="Allow output established/related"

# =====================================================
# 13. FIREWALL CONFIGURATION - FORWARD CHAIN
# =====================================================

/ip firewall filter add action=accept chain=forward connection-state=established,related comment="Accept established/related forward"
/ip firewall filter add action=accept chain=forward in-interface=wireguard-vpn out-interface=ether5-internet comment="VPN to Internet"
/ip firewall filter add action=accept chain=forward in-interface=bridge-lan comment="Allow LAN to WAN"
/ip firewall filter add action=accept chain=forward in-interface=ether5-internet out-interface=wireguard-vpn connection-state=established,related comment="Internet to VPN (return traffic)"
/ip firewall filter add action=accept chain=forward out-interface=wireguard-vpn comment="Allow all to VPN interface"
/ip firewall filter add action=drop chain=forward connection-state=invalid comment="Drop invalid"
/ip firewall filter add action=drop chain=forward comment="Drop all other forward"

# =====================================================
# 14. DNS CONFIGURATION
# =====================================================

/ip dns set allow-remote-requests=yes servers=8.8.8.8,1.1.1.1

# =====================================================
# 15. SERVICES SECURITY
# =====================================================

/ip service set telnet disabled=yes
/ip service set ftp disabled=yes
/ip service set www disabled=no
/ip service set api disabled=yes
/ip service set api-ssl disabled=yes
/ip service set ssh disabled=no port=22222
/ip service set winbox disabled=no port=8291

# =====================================================
# 16. LOGGING
# =====================================================

/system logging add topics=wireguard action=memory
/system logging add topics=firewall action=memory
